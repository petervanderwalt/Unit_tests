cmake_minimum_required(VERSION 3.20)
project(grblhal_core_tests LANGUAGES C)
include(CTest)
option(ENABLE_COVERAGE "Instrument tested core modules for coverage (GCC)" OFF)
option(ENABLE_SANITIZERS "Enable address and undefined behavior checks" OFF)
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/core/crc.c")
  message(FATAL_ERROR "Run git submodule update --init --recursive first")
endif()
function(add_core_test name module source)
  add_executable(test_${name} ${source} core/${module}.c)
  if(module STREQUAL "nuts_bolts")
    target_sources(test_${name} PRIVATE tests/support/numeric_host.c)
  endif()
  target_include_directories(test_${name} PRIVATE core tests)
  target_compile_features(test_${name} PRIVATE c_std_11)
  if(MSVC)
    target_compile_options(test_${name} PRIVATE /W4)
  else()
    target_compile_options(test_${name} PRIVATE -Wall -Wextra -Werror)
    target_link_libraries(test_${name} PRIVATE m)
    if(ENABLE_SANITIZERS)
      target_compile_options(test_${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(test_${name} PRIVATE -fsanitize=address,undefined)
    endif()
    if(ENABLE_COVERAGE)
      if(NOT CMAKE_C_COMPILER_ID STREQUAL "GNU")
        message(FATAL_ERROR "Coverage requires GCC")
      endif()
      target_compile_options(test_${name} PRIVATE --coverage -O0 -g)
      target_link_options(test_${name} PRIVATE --coverage)
    endif()
  endif()
  add_test(NAME ${name} COMMAND test_${name})
endfunction()

add_core_test(crc crc tests/test_crc.c)
add_core_test(pid pid tests/test_pid.c)
file(GLOB_RECURSE CASE_FILES CONFIGURE_DEPENDS tests/cases/*.c)
foreach(source IN LISTS CASE_FILES)
  get_filename_component(name "${source}" NAME_WE)
  get_filename_component(directory "${source}" DIRECTORY)
  get_filename_component(module "${directory}" NAME)
  add_core_test("${module}.${name}" "${module}" "${source}")
endforeach()
