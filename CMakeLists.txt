cmake_minimum_required(VERSION 3.20)
project(grblhal_core_tests LANGUAGES C)
include(CTest)
option(ENABLE_COVERAGE "Instrument tested core modules for coverage (GCC)" OFF)
option(ENABLE_SANITIZERS "Enable address and undefined behavior checks" OFF)
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/core/crc.c")
  message(FATAL_ERROR "Run git submodule update --init --recursive first")
endif()
foreach(module crc pid)
  add_executable(test_${module} tests/test_${module}.c core/${module}.c)
  target_include_directories(test_${module} PRIVATE core tests)
  target_compile_features(test_${module} PRIVATE c_std_11)
  if(MSVC)
    target_compile_options(test_${module} PRIVATE /W4)
  else()
    target_compile_options(test_${module} PRIVATE -Wall -Wextra -Werror)
    target_link_libraries(test_${module} PRIVATE m)
    if(ENABLE_SANITIZERS)
      target_compile_options(test_${module} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(test_${module} PRIVATE -fsanitize=address,undefined)
    endif()
    if(ENABLE_COVERAGE)
      if(NOT CMAKE_C_COMPILER_ID STREQUAL "GNU")
        message(FATAL_ERROR "Coverage requires GCC")
      endif()
      target_compile_options(test_${module} PRIVATE --coverage -O0 -g)
      target_link_options(test_${module} PRIVATE --coverage)
    endif()
  endif()
  add_test(NAME ${module} COMMAND test_${module})
endforeach()
