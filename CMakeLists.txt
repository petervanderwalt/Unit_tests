function(add_host_core target)
  add_library(${target} STATIC EXCLUDE_FROM_ALL tests/support/engine_host.c)
  target_link_libraries(${target} PRIVATE grbl)
  target_include_directories(${target} PUBLIC core tests)
  target_compile_features(${target} PRIVATE c_std_11)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
  target_compile_definitions(${target} PUBLIC ${ARGN})
  if(ENABLE_SANITIZERS)
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PUBLIC -fsanitize=address,undefined)
  endif()
  if(ENABLE_COVERAGE)
    target_compile_options(${target} PRIVATE --coverage -O0 -g)
    target_link_options(${target} PUBLIC --coverage)
  endif()
endfunction()

function(add_core_test name module source)
  if(MSVC AND module MATCHES "^(nuts_bolts|override|strutils|alarms|errors|encoders|crossbar|vfs|fs_ram|fs_embedded|messages|machine_limits|coolant_control|modbus|planner)$")
    message(STATUS "Skipping ${name}: core headers require GCC or Clang GNU extensions")
    return()
  endif()
  if(module IN_LIST ENGINE_MODULES)
    if(MSVC)
      message(STATUS "Skipping ${name}: host core requires GCC or Clang")
      return()
    endif()
    add_executable(test_${name} ${source})
    if(module MATCHES "^ngc_")
      target_link_libraries(test_${name} PRIVATE host_core_ngc)
    elseif(module STREQUAL "corexy")
      target_link_libraries(test_${name} PRIVATE host_core_corexy)
    else()
      target_link_libraries(test_${name} PRIVATE host_core)
    endif()
  else()
    add_executable(test_${name} ${source} core/${module}.c)
  endif()
  if(module STREQUAL "nuts_bolts")
    target_sources(test_${name} PRIVATE tests/support/numeric_host.c)
  endif()
  if(module STREQUAL "strutils")
    target_sources(test_${name} PRIVATE core/nuts_bolts.c tests/support/numeric_host.c)
  endif()
  if(module MATCHES "^(alarms|errors)$")
    target_sources(test_${name} PRIVATE tests/support/registry_host.c)
  endif()
  if(module STREQUAL "crossbar")
    target_sources(test_${name} PRIVATE tests/support/crossbar_host.c)
  endif()
  if(module MATCHES "^(machine_limits|coolant_control)$")
    target_sources(test_${name} PRIVATE tests/support/control_host.c)
  endif()
  if(module STREQUAL "coolant_control")
    target_sources(test_${name} PRIVATE tests/support/coolant_delay.c)
  endif()
  if(module STREQUAL "machine_limits")
    target_sources(test_${name} PRIVATE core/nuts_bolts.c tests/support/limits_host.c)
  endif()
  if(module MATCHES "^(vfs|fs_ram|fs_embedded)$")
    target_sources(test_${name} PRIVATE core/strutils.c core/nuts_bolts.c tests/support/numeric_host.c tests/support/filesystem_host.c)
    if(NOT module STREQUAL "vfs")
      target_sources(test_${name} PRIVATE core/vfs.c)
    endif()
  endif()
  target_include_directories(test_${name} PRIVATE core tests)
  target_compile_features(test_${name} PRIVATE c_std_11)
  if(MSVC)
    target_compile_options(test_${name} PRIVATE /W4)
  else()
    target_compile_options(test_${name} PRIVATE -Wall -Wextra -Werror)
    if(module STREQUAL "machine_limits")
      target_compile_options(test_${name} PRIVATE -ffunction-sections -fdata-sections)
      target_link_options(test_${name} PRIVATE -Wl,--gc-sections)
    endif()
    # Keep our tests strict without making upstream warnings test failures.
    set_source_files_properties(core/${module}.c PROPERTIES COMPILE_OPTIONS -Wno-error)
    target_link_libraries(test_${name} PRIVATE m)
    if(ENABLE_SANITIZERS)
      target_compile_options(test_${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(test_${name} PRIVATE -fsanitize=address,undefined)
    endif()
    if(ENABLE_COVERAGE)
      if(NOT CMAKE_C_COMPILER_ID STREQUAL "GNU")
        message(FATAL_ERROR "Coverage requires GCC")
      endif()
      target_compile_options(test_${name} PRIVATE --coverage -O0 -g)
      target_link_options(test_${name} PRIVATE --coverage)
    endif()
  endif()
  if(name STREQUAL "fs_ram.close_lifetime")
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_test(NAME ${name} COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/scripts/vfs_lifetime_regression.py
      $<BOOL:${ENABLE_SANITIZERS}> $<TARGET_FILE:test_${name}>)
    set_tests_properties(${name} PROPERTIES WILL_FAIL TRUE SKIP_RETURN_CODE 77 LABELS known_bug)
  else()
    add_test(NAME ${name} COMMAND test_${name})
  endif()
  set_tests_properties(${name} PROPERTIES TIMEOUT 15)
endfunction()
